    A proposed schema for orders, status history, table service, delays, and responses.

    order_status.sqlArchitecture example
    BEGIN;

WITH changed_order AS (
  UPDATE orders
  SET status = 'PREPARING',
      updated_at = CURRENT_TIMESTAMP
  WHERE order_id = 1042
    AND status = 'KITCHEN_RECEIVED'
  RETURNING order_id, status
)
INSERT INTO order_status_history
  (order_id, status, changed_by)
SELECT order_id, status, 'KITCHEN'
FROM changed_order;

COMMIT;
